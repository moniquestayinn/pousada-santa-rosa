const SPREADSHEET_ID = '1bPn7bSoEAggc2Ao1O_2leVdafb_2mufFVe3Pqk5BT2Y'; function doGet(e) { try { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); ensureSheets(ss); const reservas = readReservas(ss.getSheetByName('Reservas')); const suites = readSuites(ss.getSheetByName('Suites')); return jsonResponse({ success: true, reservas: reservas, suites: suites }); } catch (err) { return jsonResponse({ success: false, error: err.toString() }); } } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); ensureSheets(ss); if (data.action === 'saveReservas') { writeReservas(ss.getSheetByName('Reservas'), data.reservas || []); } else if (data.action === 'saveSuites') { writeSuites(ss.getSheetByName('Suites'), data.suites || []); } else if (data.action === 'sync') { writeReservas(ss.getSheetByName('Reservas'), data.reservas || []); writeSuites(ss.getSheetByName('Suites'), data.suites || []); } return jsonResponse({ success: true }); } catch (err) { return jsonResponse({ success: false, error: err.toString() }); } } function ensureSheets(ss) { let resSheet = ss.getSheetByName('Reservas'); if (!resSheet) { resSheet = ss.insertSheet('Reservas'); resSheet.appendRow(['id','guestName','checkIn','checkOut','suiteId','guests','reservationNumber','channel','notes','total','createdAt']); } let suiteSheet = ss.getSheetByName('Suites'); if (!suiteSheet) { suiteSheet = ss.insertSheet('Suites'); suiteSheet.appendRow(['id','name','price','capacity','description','units']); suiteSheet.appendRow(['s1','Suíte Mar',350,2,'',2]); suiteSheet.appendRow(['s2','Suíte Jardim',280,2,'',1]); } } function readReservas(sheet) { const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; const out = []; for (let i = 1; i < rows.length; i++) { const row = rows[i]; if (!row[0]) continue; out.push({ id: String(row[0]), guestName: String(row[1] || ''), checkIn: String(row[2] || ''), checkOut: String(row[3] || ''), suiteId: String(row[4] || ''), guests: Number(row[5]) || 0, reservationNumber: String(row[6] || ''), channel: String(row[7] || 'outro'), notes: String(row[8] || ''), total: Number(row[9]) || 0, createdAt: String(row[10] || '') }); } return out; } function writeReservas(sheet, reservas) { sheet.clear(); sheet.appendRow(['id','guestName','checkIn','checkOut','suiteId','guests','reservationNumber','channel','notes','total','createdAt']); reservas.forEach(function(r){ sheet.appendRow([ r.id, r.guestName, r.checkIn, r.checkOut, r.suiteId, Number(r.guests) || 0, r.reservationNumber || '', r.channel || 'outro', r.notes || '', Number(r.total) || 0, r.createdAt || new Date().toISOString() ]); }); } function readSuites(sheet) { const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; const out = []; for (let i = 1; i < rows.length; i++) { const row = rows[i]; if (!row[0]) continue; out.push({ id: String(row[0]), name: String(row[1] || ''), price: Number(row[2]) || 0, capacity: row[3] ? Number(row[3]) : null, description: String(row[4] || ''), units: row[5] ? Number(row[5]) : 1 }); } return out; } function writeSuites(sheet, suites) { sheet.clear(); sheet.appendRow(['id','name','price','capacity','description','units']); suites.forEach(function(s){ sheet.appendRow([ s.id, s.name, Number(s.price) || 0, s.capacity || '', s.description || '', (s.units !== undefined && s.units !== null) ? Number(s.units) : 1 ]); }); } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
